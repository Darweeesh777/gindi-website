      var n = $("n" + i).value === "" ? 1 : g("n" + i), o = g("o" + i);
      var ti = TYPES.map(function(t){ return t[0]; }).indexOf($("t" + i).value), s = $("s" + i).value, skOn = $("k" + i).value === "أيوه";
      var a = L === null || W === null ? 0 : Math.max(0, L * W * n - o);
      var per = L === null || W === null ? 0 : 2 * (L + W) * n, skLen = skOn ? Math.max(0, per - g("dr" + i)) : 0, sk = skLen * skH / 100;
      T[ti].area += a; T[ti].sk += sk; net += a; skAll += sk;
      $("m" + i).textContent = esc($("t" + i).value) + " · " + s + (sk ? " · وزرة " + fmt(sk, 2) + " م²" : "");
      $("g" + i).textContent = fmt(a, 2) + " م²";
      if (a || sk) A.push({name: r.querySelector(".ttl").textContent, type: T[ti].name, s: s, L: L, W: W, n: n, o: o, a: a, skLen: skLen, sk: sk});
    });

    var tot = {area: 0, sk: 0, after: 0, min: 0, avg: 0, max: 0, cost: null}, missing = false;
    T.forEach(function(t, k){
      var ctn = function(w){ return t.area + t.sk === 0 || !t.C ? 0 : up((t.area * (1 + w + wX) + t.sk * (1 + w + wX + skX)) / t.C); };
      t.after = t.area * (1 + wAvg + wX) + t.sk * (1 + wAvg + wX + skX);
      t.min = ctn(wMin); t.avg = ctn(wAvg); t.max = ctn(wMax);
      t.cost = t.avg && t.P !== null ? t.avg * t.C * t.P : null;
      t.perCtn = t.L && t.W && t.C ? Math.round(t.C / (t.L * t.W / 10000)) : null;
      t.pieces = t.L && t.W ? up(t.after / (((t.L + j / 10) * (t.W + j / 10)) / 10000)) : 0;
      if (t.area + t.sk > 0 && !t.C) missing = true;
      $("tp" + k).textContent = t.perCtn === null ? "—" : t.perCtn + " قطعة/كرتونة";
      tot.area += t.area; tot.sk += t.sk; tot.after += t.after; tot.min += t.min; tot.avg += t.avg; tot.max += t.max;
      if (t.cost !== null) tot.cost = (tot.cost || 0) + t.cost;
    });

    var cov = num($("glueCov")), gp = num($("gluePrice")), sup = num($("supplier"));
    var bags = cov && net ? up(net / cov) : null, glueCost = bags !== null && gp !== null ? bags * gp : null;
    var all = tot.cost === null ? null : tot.cost + (glueCost || 0);
    var verdict = sup === null || tot.max === 0 ? null : sup > tot.max ? "أعلى من النطاق" : sup < tot.min ? "أقل من النطاق" : "داخل النطاق";
    var verdictLong = verdict === null ? "اكتب عدد كراتين المورّد فوق" : sup > tot.max ? "أعلى من النطاق — اسأله الرقم جاي منين وبالتفصيل" : sup < tot.min ? "أقل من النطاق — اتأكد إن مفيش مكان ناقص" : "داخل النطاق المعقول";

    $("sumArea").textContent = fmt(net, 2);
    $("sumSk").textContent = fmt(skAll, 2);
    $("kCartons").textContent = fmt(tot.avg);
    $("kRange").textContent = tot.max ? "من " + tot.min + " لـ " + tot.max : "—";
    $("kArea").textContent = fmt(net + skAll, 2);
    $("kCost").textContent = all === null ? "—" : fmt(all);
    $("kCostL").textContent = all === null ? "اكتب سعر المتر لأنواع البلاط" : "التكلفة شامل اللاصق (جنيه)";
    $("kSup").textContent = verdict || "—";
    $("kSupL").textContent = verdict === null ? "اكتب كراتين المورّد فوق" : "طلب " + fmt(sup) + " كرتونة | الحسبة " + fmt(tot.avg) + " كرتونة";
    $("cmp").innerHTML = T.filter(function(t){ return t.area + t.sk > 0; }).map(function(t){
      return "<div>" + t.name + " (" + t.use + ")</div><div>" + fmt(t.avg) + " كرتونة · " + fmt(t.pieces) + " قطعة</div>";
    }).join("") +
      "<div>المساحة بعد الهالك</div><div>" + fmt(tot.after, 2) + " م²</div>" +
      "<div>تكلفة البلاط</div><div>" + (tot.cost === null ? "—" : fmt(tot.cost) + " ج") + "</div>" +
      "<div>أكياس اللاصق</div><div>" + (bags === null ? "—" : fmt(bags) + " كيس" + (glueCost === null ? "" : " · " + fmt(glueCost) + " ج")) + "</div>" +
      "<div>المقارنة مع المورّد</div><div>" + verdictLong + (verdict === null ? "" : " (الفرق عن المتوسط " + fmt(sup - tot.avg) + ")") + "</div>";
    $("check").textContent = missing ? "تنبيه: في نوع عليه مساحة ومفيش له مساحة كرتونة، اكتبها في «أنواع البلاط»." : "";

    buildReport(T, A, tot, net, skAll, bags, glueCost, all, sup, verdictLong, {wMin: wMin, wAvg: wAvg, wMax: wMax, wX: wX, j: j, skH: skH, skX: skX});
  }
