(function(){
  // Same logic as Eng_Mostafa_Elgindi_Plaster_Calculator.xlsx (sheet «الحسبة»)

  var FIELDS = [
    ["a", "المساحة (م²)"], ["th", "السُمك (مم)"], ["c", "أجزاء الأسمنت"],
    ["s", "أجزاء الرمل"], ["w", "الهالك %"], ["f", "معامل التحويل"]
  ];
  var $ = function(id){ return document.getElementById(id); };
  var fmt = function(n, d){ return n.toLocaleString("en-US", {maximumFractionDigits: d || 0}); };
  var num = function(el){ var v = parseFloat(el.value); return isFinite(v) && v >= 0 ? v : null; };
  var up = function(x){ return Math.ceil(x - 1e-9); }; // ROUNDUP(x,0) for x >= 0

  $("stages").innerHTML = STAGES.map(function(st){
    return '<div class="item"><div class="head"><div><b>' + st[1] + '</b><small id="' + st[0] + 'm">0 م³ مونة</small></div>' +
      '<div class="tot" id="' + st[0] + 'g">0 شيكارة</div></div><div class="f">' +
      FIELDS.map(function(f, k){
        // spatter shares the plaster's area and waste
        var linked = st[0] === "t" && (f[0] === "a" || f[0] === "w");
        return '<div><label for="' + st[0] + f[0] + '">' + f[1] + (linked ? ' = المحارة' : '') + '</label><input type="number" min="0" inputmode="decimal" id="' + st[0] + f[0] + '" value="' + st[k + 2] + '"' +

      }).join("") + '</div></div>';
  }).join("");

  function calc(){
    var bag = num($("bag")), dens = num($("dens")) || 0, step = num($("step"));
    $("ta").value = num($("pa")) || 0;
    $("tw").value = num($("pw")) || 0;
    var S = STAGES.map(function(st){
      var k = st[0], g = function(f){ return num($(k + f)) || 0; };
      var a = g("a"), th = g("th"), c = g("c"), s = g("s"), w = g("w") / 100, f = g("f");
      var mortar = a * th / 1000, dry = mortar * f, parts = c + s;
      var cemVol = parts ? dry * c / parts : 0, sand0 = parts ? dry * s / parts : 0;
      var kg0 = cemVol * dens, kg = kg0 * (1 + w), bags = bag ? kg / bag : 0, sand = sand0 * (1 + w);
      $(k + "m").textContent = fmt(mortar, 2) + " م³ مونة · خلطة " + fmt(c, 2) + ":" + fmt(s, 2);
      $(k + "g").textContent = fmt(bags, 2) + " شيكارة";
      return {name: st[1], a: a, th: th, c: c, s: s, w: w, f: f, mortar: mortar, dry: dry, kg: kg, bags: bags, sand: sand};
    });
    var kg = 0, bags = 0, sand = 0;
    S.forEach(function(x){ kg += x.kg; bags += x.bags; sand += x.sand; });
    var buyBags = up(bags), buySand = step ? up(sand / step) * step : sand;

    $("kBags").textContent = fmt(buyBags);
    $("kBagsL").textContent = bag ? "شيكارة " + fmt(bag, 2) + " كجم = " + fmt(buyBags * bag) + " كجم مشتراة" : "اكتب وزن الشيكارة";
    $("kSand").textContent = fmt(buySand, 2);
    $("kSandL").textContent = "م³ رمل للشراء (مقرّب لأعلى كل " + fmt(step || 0, 2) + " م³)";
    $("kCem").textContent = fmt(kg, 2);
    $("kSandC").textContent = fmt(sand, 2);
    $("cmp").innerHTML = S.map(function(x){
      return "<div>" + x.name + "</div><div>" + fmt(x.bags, 2) + " شيكارة · " + fmt(x.sand, 2) + " م³ رمل</div>";
    }).join("") + "<div>الشكاير قبل التقريب</div><div>" + fmt(bags, 2) + " شيكارة</div>" +
      S.map(function(x){
        return "<div>معدل فرد شيكارة " + x.name + "</div><div>" + (x.bags ? fmt(x.a / x.bags, 2) + " م²/شيكارة" : "—") + "</div>";
      }).join("");

    buildReport(S, kg, bags, sand, buyBags, buySand, bag, dens, step);
  }

        "<td class='hi'>" + fmt(S.reduce(function(t, x){ return t + f(x); }, 0), dg) + "</td><td>" + unit + "</td></tr>";
    };
    var inRow = function(label, f, unit){
      return "<tr><td class='l'>" + label + "</td>" + S.map(function(x){ return "<td>" + f(x) + "</td>"; }).join("") + "<td>" + unit + "</td></tr>";
    };
    $("report").innerHTML =
      "<div class='r-head'><div><h1>تقرير أسمنت ورمل المحارة والطرطشة</h1><p>حاسبة أسمنت ورمل المحارة والطرطشة · " + date + "</p></div>" +
      "<div class='r-brand'>Eng/ Mostafa Elgindi<br>Gindi Innovations</div></div>" +
      "<div class='r-kpis'><div><b>" + fmt(buyBags) + "</b><span>شيكارة أسمنت للشراء</span></div>" +
      "<div><b>" + fmt(buySand, 2) + "</b><span>م³ رمل للشراء</span></div>" +
      "<div><b>" + fmt(kg, 2) + "</b><span>كجم أسمنت شامل الهالك</span></div></div>" +
      "<p class='r-note'>وزن الشيكارة " + fmt(bag || 0, 2) + " كجم · كثافة الأسمنت " + fmt(dens) + " كجم/م³ · تقريب الرمل " + fmt(step || 0, 2) + " م³</p>" +
      "<h2>المدخلات</h2><table><thead><tr><th>المدخل</th>" + S.map(function(x){ return "<th>" + x.name + "</th>"; }).join("") + "<th>الوحدة</th></tr></thead><tbody>" +
      inRow("المساحة", function(x){ return fmt(x.a, 2); }, "م²") +
      inRow("السُمك", function(x){ return fmt(x.th, 2); }, "مم") +
