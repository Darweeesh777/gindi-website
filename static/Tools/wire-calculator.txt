    <h2>منطق الحسبة</h2>
    <dl>
      <dt>اللينية</dt><dd>الدايرة اللي بتطلع من قاطع في اللوحة وتخدم مجموعة نقاط أو جهاز واحد.</dd>
      <dt>عدد الأسلاك</dt><dd>كل لينية = 3 أسلاك: فاز + نيوترال + أرضي، فالطول بيتضرب في 3.</dd>
      <dt>مفتاح إنارة</dt><dd>السلك اللي نازل من السقف للمفتاح، وبيتحسب مع كل غرفة: سلكين (المفتاح بيقطع الفاز بس) بطول ثابت 5 متر لكل مفتاح، مقاس 1.5 مم.</dd>
      <dt>البرايز</dt><dd>كل غرفة ليها لينية بريزة واحدة من اللوحة لأول بريزة، وباقي برايز الغرفة تمديدات منها. الطول = طول اللينية + (عدد البرايز − 1) × طول التمديدة، وبيتضرب في 3 أسلاك (فاز + نيوترال + أرضي) بمقاس 2 مم، أو سلكين بس لو البرايز مش متأرضة.</dd>
      <dt>المقاسات</dt><dd>إنارة 1.5 مم، برايز 2 مم، غسالات وتكييفات 4 مم، سخان فوري وفرن 6 مم.</dd>
      <dt>المساحة</dt><dd>الإنارة ومفاتيح الإنارة والبرايز بيتضرب طولها في مساحة الشقة ÷ المساحة المرجعية. لو مساحة الشقة فاضية بيتحسب من غير تحجيم. الأجهزة لينيات مستقلة ومش بتتأثر بالمساحة.</dd>
      <dt>الهالك والنطاق</dt><dd>الهالك 10% أدنى و12% متوسط و15% أعلى، وهامش ±15% للأطوال بيدي النطاق الأدنى والأعلى للفات.</dd>
      <dt>الغرف والأجهزة</dt><dd>كل غرفة = لينية إنارة + مفتاح إنارة + لينية برايز بتمديداتها (تقدر تزود عدد المفاتيح لو الغرفة فيها أكتر من مفتاح). كل خانة تكييف أو سخان فوري = جهاز واحد. دوس «+ إضافة» وهتتفتح خانة جديدة بنفس المواصفات.</dd>

  function calc(){
    var area = num($("area")), ref = num($("ref"));
    var roll = num($("roll")) || 0;
    var margin = (num($("margin")) || 0) / 100;
    var wmin = (num($("wmin")) || 0) / 100, wavg = (num($("wavg")) || 0) / 100, wmax = (num($("wmax")) || 0) / 100;
    var factor = (area === null || !ref) ? 1 : area / ref;

    var bySize = {}; SIZES.forEach(function(s){ bySize[s] = 0; });
    var sum = 0, P = [];
    box.querySelectorAll(".item").forEach(function(r){
      var i = r.dataset.row, cEl = $("c" + i);
      var c = cEl ? (num(cEl) || 0) : 1, l = num($("l" + i)) || 0, w = num($("w" + i)) || 0;
      var f = $("a" + i).value === "نعم" ? factor : 1;
      var g = c * l * w * f, nm = r.querySelector(".ttl").textContent, sz = parseFloat($("s" + i).value);
      var nmEl = $("n" + i); if (nmEl && nmEl.value.trim()) nm = nmEl.value.trim();
      bySize[sz] += g;
      P.push({name: nm + ($("k" + i) ? " — إنارة" : ""), size: sz, qty: cEl ? c : 1, len: g});
      var kEl = $("k" + i);
      if (kEl) { var sw = (num(kEl) || 0) * SWITCH.len * SWITCH.wires * f; bySize[SWITCH.size] += sw; g += sw;
        P.push({name: nm + " — مفتاح إنارة", size: SWITCH.size, qty: num(kEl) || 0, len: sw}); }
      var nEl = $("sn" + i);
      if (nEl) {
        var n = num(nEl) || 0;
        var so = n > 0 ? ((num($("sl" + i)) || 0) + (n - 1) * (num($("se" + i)) || 0)) * parseFloat($("se_e" + i).value) * f : 0;
        bySize[SOCKET.size] += so; g += so;
        P.push({name: nm + " — برايز" + ($("se_e" + i).value === "2" ? " (من غير أرضي)" : ""), size: SOCKET.size, qty: n, len: so});
      }
      sum += g;
      $("g" + i).textContent = fmt(g, 1) + " م";
    });
    $("sumLen").textContent = fmt(sum, 1);

    var T = {len:0, after:0, min:0, avg:0, max:0, cost:0}, anyPrice = false, S = [];
    SIZES.forEach(function(s, i){
      var L = bySize[s], D = L * (1 + wavg);
      var mn = roll ? up(L * (1 - margin) * (1 + wmin) / roll) : 0;
      var av = roll ? up(D / roll) : 0;
      var mx = roll ? up(L * (1 + margin) * (1 + wmax) / roll) : 0;
      var p = num($("p" + i));
      $("L" + i).textContent = fmt(L, 1); $("D" + i).textContent = fmt(D, 1);
      $("E" + i).textContent = mn; $("F" + i).textContent = av; $("G" + i).textContent = mx;
      $("I" + i).textContent = p === null ? "" : fmt(av * p);
      if (p !== null) { anyPrice = true; T.cost += av * p; }
      S.push({size: s, L: L, D: D, mn: mn, av: av, mx: mx, cost: p === null ? null : av * p});
      T.len += L; T.after += D; T.min += mn; T.avg += av; T.max += mx;
    });
    $("tLen").textContent = fmt(T.len, 1); $("tAfter").textContent = fmt(T.after, 1);
    $("tMin").textContent = T.min; $("tAvg").textContent = T.avg; $("tMax").textContent = T.max;
    $("tCost").textContent = anyPrice ? fmt(T.cost) + " ج" : "";

    var range = T.max === 0 ? "—" : T.min + " – " + T.max;
    $("kTotal").textContent = T.avg;
    $("kRange").textContent = range;
    $("kLen").textContent = fmt(T.len, 1);
    $("kLenL").textContent = area === null ? "متر قبل الهالك" : "متر قبل الهالك — شقة " + fmt(area, 1) + " م²";

    var e = num($("elec")), v = $("verdict");
    $("cElec").textContent = e === null ? "—" : fmt(e, 1);
    $("cRange").textContent = T.max === 0 ? "—" : range + " لفة";
    $("cDiff").textContent = e === null ? "—" : (e - T.avg > 0 ? "+" : "") + fmt(e - T.avg, 1) + " لفة";
    if (e === null) { v.className = "verdict"; v.textContent = "اكتب عدد اللفات اللي الكهربائي طلبها عشان تقارن."; }
    else if (e > T.max) { v.className = "verdict bad"; v.textContent = "أعلى من النطاق: اسأله الرقم جاي منين وبالتفصيل."; }
    else if (e < T.min) { v.className = "verdict bad"; v.textContent = "أقل من النطاق: اتأكد إن مفيش بند ناقص."; }
    else { v.className = "verdict ok"; v.textContent = "داخل النطاق المعقول ✓"; }

    buildReport(P, S, T, anyPrice, area, e, v.textContent);
  }

  function esc(t){ return String(t).replace(/[&<>"]/g, function(ch){ return {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[ch]; }); }

      "<p class='r-note'>طول اللفة " + fmt(num($("roll")) || 0) + " م · الهالك " + fmt(num($("wmin")) || 0, 1) + "% / " + fmt(num($("wavg")) || 0, 1) + "% / " + fmt(num($("wmax")) || 0, 1) + "% · هامش الأطوال ±" + fmt(num($("margin")) || 0, 1) + "% · اللفات متقرّبة لأعلى لكل مقاس." + (anyPrice ? " · الأسعار من لستة السويدي (مجدول) " + PRICE_DATE + "، وممكن تتغير." : "") + "</p>" +
      "<h2>تفصيل البنود</h2><table><thead><tr><th>البند</th><th>المقاس</th><th>العدد</th><th>الطول (م)</th></tr></thead><tbody>" + parts + "</tbody></table>" +
      (e !== null ? "<h2>المقارنة مع طلب الكهربائي</h2><p class='r-cmp'>الكهربائي طلب <b>" + fmt(e, 1) + "</b> لفة · " + esc(verdict) + "</p>" : "") +
      "<p class='r-warn'>الحسبة تقريبية وبتتغير حسب توزيع اللوحة وعدد النقاط وأطوال المسارات والأحمال، ولازم يعتمدها مهندس كهرباء مختص.</p>" +

  }
